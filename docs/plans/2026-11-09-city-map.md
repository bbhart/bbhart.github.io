```python
# /// script
# dependencies = ["pyyaml"]
# ///
"""List post location tags not matched by _data/cities.yml, most used first."""
import collections
import pathlib
import re
import sys

import yaml

root = pathlib.Path(sys.argv[1])
path = root / "_data/cities.yml"
cities = (yaml.safe_load(path.read_text()) if path.exists() else None) or {}
known = set(cities)
for info in cities.values():
    known.update(info.get("aliases") or [])

counts, kinds = collections.Counter(), collections.defaultdict(set)
matched = set()
for post in (root / "_posts").glob("*.m*"):
    m = re.match(r"---\n(.*?)\n---", post.read_text(), re.S)
    if not m:
        continue
    fm = yaml.safe_load(m.group(1)) or {}
    for field in ("locations", "locations_visited"):
        locs = fm.get(field) or []
        for loc in locs if isinstance(locs, list) else [locs]:
            name = loc.get("name") if isinstance(loc, dict) else loc
            if not name:
                continue
            if name in known:
                matched.add(name)
                continue
            counts[name] += 1
            kinds[name].add(loc.get("type", "-") if isinstance(loc, dict) else "-")

for name, count in counts.most_common():
    print(count, name, sorted(kinds[name]))
print("unused keys or aliases:", sorted(known - matched))
```

```js
// Shared by the home city map and the /trips/ route map: the Leaflet setup,
// vector basemap, pins, and the numbered bubbles overlapping pins fold into.
// items: [{ id, at: [lat, lng], label (HTML), title, name }]
//   name: the short name used in bubble tooltips.
// opts: { onSelect: function (id), maxZoom }
(function () {
  window.createAtlas = function (el, items, opts) {
    var narrow = window.matchMedia("(max-width: 720px)");
    // Touch screens get a 44px tap target around each pin (the dot itself is
    // drawn by CSS), and pins merge sooner so those targets don't overlap.
    var touch = window.matchMedia("(pointer: coarse)").matches;
    var HIT = touch ? 44 : 18;
    var GROUP_PX = HIT + 4;
    var CLUSTER = touch ? 44 : 30;
    // On phones, keep fitted pins clear of the zoom buttons and attribution
    // in the bottom-right corner, where a pin can't be tapped.
    var PHONE_BR = [70, 110];

    var map = L.map(el, {
      zoomControl: false,
      scrollWheelZoom: false,
      minZoom: narrow.matches ? 0.5 : 2,
      maxZoom: opts.maxZoom || 6,
      zoomSnap: 0.25,
      maxBounds: [[-60, -200], [85, 200]],
      attributionControl: false
    });
    L.control.zoom({ position: "bottomright" }).addTo(map);
    L.control.attribution({ prefix: false, position: "bottomright" })
      .addAttribution('Map data: <a href="https://www.naturalearthdata.com/">Natural Earth</a>')
      .addTo(map);

    // Vector basemap: Natural Earth 1:50m countries, colored by the site's CSS
    // tokens (see .land in _home.scss), so it follows light/dark mode with no
    // tile service or API key.
    map.createPane("land").style.zIndex = 200;
    fetch("https://cdn.jsdelivr.net/npm/world-atlas@2/countries-50m.json")
      .then(function (r) { return r.json(); })
      .then(function (topo) {
        var world = topojson.feature(topo, topo.objects.countries);
        // Drop Antarctica, and unwrap rings that straddle the antimeridian
        // (Russia, Fiji, the Aleutians) so they don't smear across the map.
        world.features = world.features.filter(function (f) { return f.id !== "010"; });
        world.features.forEach(function (f) {
          var polys = f.geometry.type === "Polygon" ? [f.geometry.coordinates] : f.geometry.coordinates;
          polys.forEach(function (poly) {
            poly.forEach(function (ring) {
              var east = ring.some(function (c) { return c[0] > 160; });
              var west = ring.some(function (c) { return c[0] < -160; });
              if (east && west) ring.forEach(function (c) { if (c[0] < 0) c[0] += 360; });
            });
          });
        });
        L.geoJSON(world, {
          pane: "land",
          interactive: false,
          style: { className: "land", weight: 0.6 }
        }).addTo(map);
      });

    var pins = {};
    var active = null;

    function pinIcon(item, on) {
      return L.divIcon({
        className: "trip-pin" + (on ? " is-active" : ""),
        html: '<span class="trip-pin__dot"></span><span class="trip-pin__label">' + item.label + "</span>",
        iconSize: [HIT, HIT],
        iconAnchor: [HIT / 2, HIT / 2]
      });
    }

    // Set the view before adding markers: Leaflet defers layers added to a map
    // with no view, and regroup() can't remove a marker that isn't placed yet.
    var bounds = L.latLngBounds(items.map(function (i) { return i.at; }));
    function home() {
      // Leave room on the left for the intro panel on wide screens.
      map.fitBounds(bounds, narrow.matches
        ? { paddingTopLeft: [20, 20], paddingBottomRight: [PHONE_BR[0] - 30, PHONE_BR[1] - 50], maxZoom: 4 }
        : { paddingTopLeft: [400, 50], paddingBottomRight: [50, 50], maxZoom: 4 });
    }
    home();

    items.forEach(function (item) {
      var m = L.marker(item.at, { icon: pinIcon(item, false), title: item.title, keyboard: true, riseOnHover: true })
        .addTo(map)
        .on("click", function () { opts.onSelect(item.id); });
      pins[item.id] = { marker: m, item: item };
    });

    // Pins that land on top of each other (London and Paris at world zoom)
    // fold into a numbered bubble; tapping it zooms in until they separate.
    var groups = L.layerGroup().addTo(map);
    // Tap targets are squares, so compare the larger of the x and y gaps:
    // straight-line distance lets two squares overlap on the diagonal.
    function tooClose(a, b) {
      return Math.max(Math.abs(a.x - b.x), Math.abs(a.y - b.y)) < GROUP_PX;
    }
    function point(p) { return map.latLngToContainerPoint(p.item.at); }
    function regroup() {
      groups.clearLayers();
      var left = Object.keys(pins).map(function (k) { return pins[k]; });
      left.forEach(function (p) { if (!map.hasLayer(p.marker)) p.marker.addTo(map); });
      // A bubble at maximum zoom could never open, so leave those pins separate.
      if (map.getZoom() >= map.getMaxZoom()) return;
      while (left.length) {
        var head = left.shift();
        var hp = point(head);
        var near = left.filter(function (p) { return tooClose(hp, point(p)); });
        if (!near.length) continue;
        left = left.filter(function (p) { return near.indexOf(p) < 0; });
        var members = [head].concat(near);
        // The bubble sits at the group's center, which can land on a pin that
        // wasn't close to the first member; pull those in until nothing is.
        for (var grew = true; grew;) {
          var c = map.latLngToContainerPoint(L.latLngBounds(members.map(function (p) { return p.item.at; })).getCenter());
          var more = left.filter(function (p) { return tooClose(c, point(p)); });
          grew = more.length > 0;
          members = members.concat(more);
          left = left.filter(function (p) { return more.indexOf(p) < 0; });
        }
        if (members.some(function (p) { return p.item.id === active; })) continue;
        members.forEach(function (p) { map.removeLayer(p.marker); });
        addGroup(members);
      }
    }

    function addGroup(members) {
      var b = L.latLngBounds(members.map(function (p) { return p.item.at; }));
      L.marker(b.getCenter(), {
        icon: L.divIcon({ className: "trip-cluster", html: "<span>" + members.length + "</span>", iconSize: [CLUSTER, CLUSTER] }),
        title: members.map(function (p) { return p.item.name; }).join(", "),
        keyboard: true,
        zIndexOffset: 500
      }).on("click", function () {
        var fitOpts = narrow.matches
          ? { paddingTopLeft: [50, 50], paddingBottomRight: PHONE_BR, duration: 0.6 }
          : { paddingTopLeft: [420, 80], paddingBottomRight: [80, 80], duration: 0.6 };
        // Always zoom in at least a step, or a group that re-forms at the
        // fitted zoom would never come apart.
        var fitZoom = map.getBoundsZoom(b, false, L.point(narrow.matches ? 140 : 500, 160));
        if (fitZoom > map.getZoom() + 0.5) {
          map.flyToBounds(b, fitOpts);
        } else {
          map.flyTo(b.getCenter(), Math.min(map.getZoom() + 1.5, map.getMaxZoom()), { duration: 0.6 });
        }
      }).addTo(groups);
    }
    map.on("zoomend", regroup);
    regroup();

    function setActive(id) {
      if (active && pins[active]) pins[active].marker.setIcon(pinIcon(pins[active].item, false));
      active = id;
      if (id && pins[id]) pins[id].marker.setIcon(pinIcon(pins[id].item, true));
    }

    return {
      map: map,
      narrow: narrow,
      PHONE_BR: PHONE_BR,
      pins: pins,
      home: home,
      setActive: setActive,
      active: function () { return active; }
    };
  };
})();
```

**Step 3: Replace `assets/trip-map.js`**

```js
// /trips/ route map. Reads the recap list from #trip-data (rendered by
// _layouts/trips.html from _data/trips.yml) and draws one pin per trip.
// Selecting a pin shows the trip card and traces its route.
(function () {
  var el = document.getElementById("trip-map");
  var dataEl = document.getElementById("trip-data");
  if (!el || !dataEl || !window.L || !window.createAtlas) return;

  var trips = JSON.parse(dataEl.textContent).filter(function (t) { return t.pin; });
  var bySlug = {};
  trips.forEach(function (t) { bySlug[t.slug] = t; });

  var atlas = window.createAtlas(el, trips.map(function (t) {
    return { id: t.slug, at: t.pin, label: t.label + " <b>" + t.year + "</b>", title: t.title, name: t.label };
  }), { onSelect: select });
  var map = atlas.map;
  var narrow = atlas.narrow;

  var route = L.layerGroup().addTo(map);
  var card = document.getElementById("trip-card");
  var box = el.closest(".atlas");

  function select(slug) {
    var trip = bySlug[slug];
    if (!trip) return;
    atlas.setActive(slug);

    route.clearLayers();
    var pts = trip.stops.map(function (s) { return [s[1], s[2]]; });
    for (var i = 1; i < pts.length; i++) {
      L.polyline(arc(pts[i - 1], pts[i]), { className: "trip-route", weight: 2, dashArray: "4 6", interactive: false }).addTo(route);
    }
    trip.stops.forEach(function (s) {
      L.circleMarker([s[1], s[2]], { className: "trip-stop", radius: 4, weight: 2, interactive: true })
        .bindTooltip(s[0], { direction: "top", offset: [0, -4], className: "trip-tip" })
        .addTo(route);
    });

    var fit = pts.length ? L.latLngBounds(pts) : L.latLngBounds([trip.pin]);
    map.flyToBounds(fit, {
      paddingTopLeft: narrow.matches ? [30, 30] : [420, 60],
      paddingBottomRight: narrow.matches ? atlas.PHONE_BR : [60, 60],
      maxZoom: 6,
      duration: 0.8
    });

    showCard(trip);
  }

  // A gentle curve between two stops (quadratic Bézier bowed to one side),
  // which reads as travel rather than a straight line over land.
  function arc(a, b) {
    var dLat = b[0] - a[0], dLng = b[1] - a[1];
    var ctrl = [(a[0] + b[0]) / 2 + dLng * 0.18, (a[1] + b[1]) / 2 - dLat * 0.18];
    var out = [];
    for (var s = 0; s <= 20; s++) {
      var u = s / 20, v = 1 - u;
      out.push([v * v * a[0] + 2 * v * u * ctrl[0] + u * u * b[0], v * v * a[1] + 2 * v * u * ctrl[1] + u * u * b[1]]);
    }
    return out;
  }

  function showCard(t) {
    var img = card.querySelector("img");
    card.querySelector(".atlas__card-link").href = t.url;
    card.querySelector(".kicker").textContent = t.year;
    card.querySelector("h2").textContent = t.title;
    card.querySelector(".atlas__card-sub").textContent = t.subtitle || "";
    if (t.image) { img.src = t.image; img.hidden = false; } else { img.hidden = true; }
    card.hidden = false;
    box.classList.add("is-selected");
  }

  function clear() {
    atlas.setActive(null);
    route.clearLayers();
    card.hidden = true;
    box.classList.remove("is-selected");
    atlas.home();
  }

  card.querySelector(".atlas__close").addEventListener("click", clear);
  document.addEventListener("keydown", function (e) { if (e.key === "Escape" && atlas.active()) clear(); });

  // Hovering a trip in the grid below lifts its pin on the map.
  document.querySelectorAll(".trip-card[data-trip]").forEach(function (a) {
    var slug = a.getAttribute("data-trip");
    if (!atlas.pins[slug]) return;
    // getElement() is null while the pin is folded into a group.
    a.addEventListener("mouseenter", function () { var e = atlas.pins[slug].marker.getElement(); if (e) e.classList.add("is-hover"); });
    a.addEventListener("mouseleave", function () { var e = atlas.pins[slug].marker.getElement(); if (e) e.classList.remove("is-hover"); });
  });
})();
```

```liquid
{%- comment -%}
  Builds the home page city index from _data/cities.yml and post locations.
  Sets city_rows (array of JSON strings, one per city with posts) and city_count.
  One card per trip: see docs/plans/2026-11-09-city-map-design.md.
{%- endcomment -%}
{%- assign nothing = "" | split: "" -%}
{%- assign loc_fields = "locations,locations_visited" | split: "," -%}
{%- assign city_rows = nothing -%}
{%- for entry in site.data.cities -%}
  {%- assign names = entry[1].aliases | default: nothing | push: entry[0] -%}
  {%- assign hits = nothing -%}
  {%- assign hit_keys = nothing -%}
  {%- for post in site.posts -%}
    {%- assign hit = false -%}
    {%- for field in loc_fields -%}
      {%- for loc in post[field] -%}
        {%- assign loc_name = loc.name | default: loc -%}
        {%- if names contains loc_name -%}{%- assign hit = true -%}{%- endif -%}
      {%- endfor -%}
    {%- endfor -%}
    {%- if hit -%}
      {%- assign key = post.rollup_key | default: site.data.trips[post.slug].rollup_key | default: post.url -%}
      {%- assign hits = hits | push: post -%}
      {%- assign hit_keys = hit_keys | push: key -%}
    {%- endif -%}
  {%- endfor -%}
  {%- if hits.size == 0 -%}{%- continue -%}{%- endif -%}

  {%- assign hits_oldest = hits | reverse -%}
  {%- assign keys_oldest = hit_keys | reverse -%}
  {%- assign seen = nothing -%}
  {%- assign cards = nothing -%}
  {%- for key in hit_keys -%}
    {%- if seen contains key -%}{%- continue -%}{%- endif -%}
    {%- assign seen = seen | push: key -%}

    {%- assign chosen = nil -%}
    {%- assign recap_hit = nil -%}
    {%- for candidate in hits_oldest -%}
      {%- if keys_oldest[forloop.index0] != key -%}{%- continue -%}{%- endif -%}
      {%- if candidate.categories contains "rollup" -%}
        {%- assign recap_hit = candidate -%}
      {%- elsif chosen == nil -%}
        {%- assign chosen = candidate -%}
      {%- endif -%}
    {%- endfor -%}
    {%- if chosen == nil -%}{%- assign chosen = recap_hit -%}{%- endif -%}

    {%- assign trip_slug = nil -%}
    {%- assign trip = nil -%}
    {%- for t in site.data.trips -%}
      {%- if t[1].rollup_key == key -%}
        {%- assign trip_slug = t[0] -%}
        {%- assign trip = t[1] -%}
        {%- break -%}
      {%- endif -%}
    {%- endfor -%}

    {%- assign recap_image = nil -%}
    {%- if trip_slug -%}
      {%- assign trip_recap = site.posts | where: "slug", trip_slug | first -%}
      {%- assign recap_image = trip_recap.background -%}
    {%- endif -%}
    {%- assign image = chosen.background | default: recap_image -%}

    {%- if trip -%}
      {%- capture label -%}{{ trip.label }} {{ trip.year }}{%- endcapture -%}
    {%- else -%}
      {%- assign label = chosen.title -%}
    {%- endif -%}

    {%- capture card -%}{"url":{{ chosen.url | relative_url | jsonify }},"image":{% if image %}{{ image | relative_url | jsonify }}{% else %}null{% endif %},"date":{{ chosen.date | date: "%Y-%m-%d" | jsonify }},"dateLabel":{{ chosen.date | date: "%b %-d, %Y" | jsonify }},"trip":{{ label | jsonify }}}{%- endcapture -%}
    {%- assign cards = cards | push: card -%}
  {%- endfor -%}

  {%- capture row -%}{"name":{{ entry[0] | jsonify }},"coords":{{ entry[1].coords | jsonify }},"cards":[{{ cards | join: "," }}]}{%- endcapture -%}
  {%- assign city_rows = city_rows | push: row -%}
{%- endfor -%}
{%- assign city_count = city_rows.size -%}
```

**Step 4: Create `assets/city-map.js`**

```js
// Home page city map. Reads #city-data (rendered by _includes/city_index.html)
// and draws one pin per city. Selecting a pin lists one card per trip there.
(function () {
  var el = document.getElementById("city-map");
  var dataEl = document.getElementById("city-data");
  if (!el || !dataEl || !window.L || !window.createAtlas) return;

  var cities = JSON.parse(dataEl.textContent);
  var byName = {};
  cities.forEach(function (c) { byName[c.name] = c; });

  // Zoom further than the route map: towns a few kilometres apart can't
  // separate at zoom 6, and would be stuck inside a bubble.
  var atlas = window.createAtlas(el, cities.map(function (c) {
    return { id: c.name, at: c.coords, label: c.name, title: c.name, name: c.name };
  }), { onSelect: select, maxZoom: 12 });

  var panel = document.getElementById("city-card");
  var list = panel.querySelector(".city-cards");
  var box = el.closest(".atlas");

  function text(tag, className, value) {
    var node = document.createElement(tag);
    node.className = className;
    node.textContent = value;
    return node;
  }

  function cardItem(card) {
    var link = document.createElement("a");
    link.className = "trip-card";
    link.href = card.url;
    if (card.image) {
      var img = document.createElement("img");
      img.alt = "";
      img.src = card.image;
      link.appendChild(img);
    } else {
      var blank = text("span", "trip-card__blank", card.trip);
      blank.setAttribute("aria-hidden", "true");
      link.appendChild(blank);
    }
    link.appendChild(text("span", "trip-card__year", card.dateLabel));
    link.appendChild(text("span", "trip-card__title", card.trip));
    var li = document.createElement("li");
    li.appendChild(link);
    return li;
  }

  function select(name) {
    var city = byName[name];
    if (!city) return;
    atlas.setActive(name);
    atlas.map.panInside(city.coords, atlas.narrow.matches
      ? { padding: [30, 30] }
      : { paddingTopLeft: [420, 60], paddingBottomRight: [60, 60] });

    panel.querySelector("h2").textContent = city.name;
    var newest = city.cards.slice().sort(function (a, b) { return a.date < b.date ? 1 : -1; });
    list.replaceChildren.apply(list, newest.map(cardItem));
    panel.hidden = false;
    panel.scrollTop = 0;
    box.classList.add("is-selected");
    // On phones the panel sits below the map, possibly off screen.
    if (atlas.narrow.matches) panel.scrollIntoView({ behavior: "smooth", block: "nearest" });
  }

  function clear() {
    atlas.setActive(null);
    panel.hidden = true;
    box.classList.remove("is-selected");
  }

  panel.querySelector(".atlas__close").addEventListener("click", clear);
  document.addEventListener("keydown", function (e) { if (e.key === "Escape" && atlas.active()) clear(); });
})();
```
